import express from "express";
import dotenv from "dotenv";
import { GoogleGenAI } from "@google/genai";

dotenv.config();

const app = express();
const PORT = process.env.PORT || 3000;

const ai = new GoogleGenAI({
  apiKey: process.env.GEMINI_API_KEY
});

app.use(express.json());
app.use(express.static("public"));

const SYSTEM_INSTRUCTION = `
You are EduGenie, an intelligent and friendly learning assistant.

Your goals:
1. Help students understand concepts clearly.
2. Explain difficult topics step-by-step.
3. Adapt explanations to the student's level.
4. Give examples when useful.
5. Ask short follow-up questions when appropriate.
6. Help with homework without simply doing everything for the student.
7. Create quizzes, flashcards, summaries and study plans when requested.
8. Encourage students to reason through problems.
9. If a student makes a mistake, explain the mistake kindly.
10. Never pretend to know something you do not know.

For mathematical problems:
- Show the reasoning step-by-step.
- Give the final answer clearly.

For learning requests:
- Use headings and bullet points where helpful.
- Keep explanations understandable.
- Use Markdown formatting.

You are an educational assistant, not a replacement for a teacher.
`;

app.post("/api/chat", async (req, res) => {
  try {
    const { message, history = [] } = req.body;

    if (!message || typeof message !== "string") {
      return res.status(400).json({
        error: "Please provide a message."
      });
    }

    const contents = [
      ...history.map(item => ({
        role: item.role === "assistant" ? "model" : "user",
        parts: [
          {
            text: String(item.text)
          }
        ]
      })),
      {
        role: "user",
        parts: [
          {
            text: message
          }
        ]
      }
    ];

    const response = await ai.models.generateContent({
      model: "gemini-3.8-flash",
      contents,
      config: {
        systemInstruction: SYSTEM_INSTRUCTION,
        temperature: 0.7,
        maxOutputTokens: 1500
      }
    });

    res.json({
      answer: response.text
    });

  } catch (error) {
    console.error("Gemini error:", error);

    res.status(500).json({
      error: "EduGenie could not generate a response."
    });
  }
});

app.listen(PORT, () => {
  console.log(`EduGenie running at http://localhost:${PORT}`);
});
